   - 🌟중요(브라우저 유지): 반드시 detach 옵션을 사용하세요:
     ```python
     from selenium.webdriver.chrome.options import Options
     options = Options()
     options.add_experimental_option('detach', True)  # 스크립트 종료 후 브라우저 유지
     driver = webdriver.Chrome(service=Service(ChromeDriverManager().install()), options=options)
     ```
   - detach=True가 없으면 스크립트가 끝날 때 Chrome이 자동으로 닫힙니다.
   - 다음 스텝에서 이미 열린 브라우저를 재사용하려면 driver 객체를 유지하거나,
     Chrome 실행 시 --remote-debugging-port=9222 옵션 추가 후 connect로 재연결하세요.
   - 🌟중요(요소 찾기 핵심 규칙 — 모든 웹 환경 공통):
     ① XPath로 텍스트 검색 시 반드시 script/style 태그 제외하세요:
        잘못된 예: //*[contains(text(), 'RPA')]  ← <script> 태그의 JS 문자열도 매칭되어 클릭 실패!
        올바른 예: //*[not(self::script)][not(self::style)][contains(text(), 'RPA')]
     ② 버튼 클릭 후 나타나는 드롭다운/서브메뉴는 visibility_of_element_located를 사용하세요:
        잘못된 예: wait.until(EC.presence_of_element_located(...))  ← hidden 요소도 즉시 반환!
        올바른 예: wait.until(EC.visibility_of_element_located(...))  ← 실제 화면에 보일 때까지 대기
     ③ 비인터랙티브 텍스트 요소(<span>, <div>, <li> 등) 클릭 시 JavaScript click을 사용하세요:
        element = wait.until(EC.visibility_of_element_located((By.XPATH, '//*[not(self::script)][not(self::style)][contains(text(), "메뉴명")]')))
        driver.execute_script("arguments[0].click()", element)  # JS click으로 안정적 클릭
     ④ <button>, <a>, <input> 같은 인터랙티브 요소는 element_to_be_clickable + .click()을 사용하세요:
        element = wait.until(EC.element_to_be_clickable((By.XPATH, '//button[contains(text(), "확인")]')))
        element.click()
     ⑤ 로케이터 안정성 우선순위: id속성 > title속성 > aria-label > 텍스트내용
        - title 속성: 사이드바·트리 메뉴에서 필수 (텍스트가 <span> 자식에 있어 text() XPath가 실패하는 구조)
          //*[@title="메뉴명"]   ← zTree, axBoot 등 SPA 프레임워크 메뉴에서 가장 안정적
     ⑥ 웹 요소 클릭 표준 함수: 모든 웹 자동화 코드에 아래 find_and_click()을 항상 포함하세요.
        off-canvas(화면 밖 x<0) 요소도 JS click으로 자동 처리되므로 클릭 실패가 없습니다:
        ```python
        def find_and_click(driver, locators, timeout=10):
            from selenium.webdriver.support.ui import WebDriverWait
            from selenium.webdriver.support import expected_conditions as EC
            from selenium.webdriver.common.by import By
            last_err = None
            for strategy, value in locators:
                try:
                    by_map = {'id': By.ID, 'css': By.CSS_SELECTOR, 'xpath': By.XPATH}
                    if strategy == 'title':
                        by, val = By.XPATH, f'//*[@title="{value}"]'
                    elif strategy == 'text':
                        by, val = By.XPATH, f'//*[not(self::script)][not(self::style)][normalize-space(.)="{value}"]'
                    else:
                        by, val = by_map.get(strategy, By.XPATH), value
                    el = WebDriverWait(driver, timeout).until(EC.presence_of_element_located((by, val)))
                    rect = driver.execute_script(
                        'var r=arguments[0].getBoundingClientRect(); return {x:r.x,y:r.y};', el)
                    if rect['x'] < 0 or rect['y'] < 0:
                        driver.execute_script('arguments[0].click()', el)
                    else:
                        try: el.click()
                        except Exception: driver.execute_script('arguments[0].click()', el)
                    return el
                except Exception as e:
                    last_err = e
            raise Exception(f'클릭 실패: {locators} / {last_err}')
        ```
        사용 예: find_and_click(driver, [('title', '사용자코드 관리'), ('text', '사용자코드 관리')])

### 부모 윈도우: DBeaver

### 코드 생성 컨텍스트:
> **⚠️ 권한 주의사항**: 대상 앱이 관리자 권한으로 실행 중이면:
> - `click()` (WM 메시지): 권한 오류 발생
> - `click_input()` (마우스 시뮬레이션): 권한 오류 발생
> - `pyautogui.click(x, y)` (입력 시뮬레이션): **정상 동작** (UIPI 우회)
> - 단, pyautogui는 **실행 시점의 요소 좌표**를 동적으로 가져와야 정확함

```python
import ctypes
import ctypes.wintypes
import time
import pyautogui
from pywinauto import Application

# ★ DPI Awareness 설정 (pywinauto 좌표와 pyautogui 좌표 일치 보장)
# 이 스크립트를 호출한 프로세스와 동일한 DPI 모드로 맞춤
try:
    ctypes.windll.user32.SetProcessDpiAwarenessContext(ctypes.c_void_p(-4))
except Exception:
    try:
        ctypes.windll.shcore.SetProcessDpiAwareness(2)  # PROCESS_PER_MONITOR_DPI_AWARE
    except Exception:
        pass

pyautogui.FAILSAFE = False  # 모서리 이동에 의한 abort 방지

app = Application(backend="uia").connect(title="DBeaver", timeout=10)
win = app.window(title="DBeaver")
element = win.child_window(class_name="SWT_Window0", control_type="Pane", found_index=0)

# 실행 시점의 요소 위치를 동적으로 계산
rect = element.rectangle()
center_x = (rect.left + rect.right) // 2
center_y = (rect.top + rect.bottom) // 2
print(f'요소 위치: ({center_x}, {center_y})  크기: {rect.width()}x{rect.height()}')

# 대상 창을 최상위로 가져오기 (관리자 창이면 일부 실패할 수 있음)
user32 = ctypes.windll.user32
hwnd = win.handle
user32.ShowWindow(hwnd, 9)     # SW_RESTORE
user32.BringWindowToTop(hwnd)
try:
    user32.SetForegroundWindow(hwnd)
except Exception:
    pass
time.sleep(0.5)  # 창 전환 대기

# 클릭 좌표에 실제로 있는 창 확인
pt = ctypes.wintypes.POINT(center_x, center_y)
hwnd_at_pt = user32.WindowFromPoint(pt)
buf = ctypes.create_unicode_buffer(256)
user32.GetWindowTextW(hwnd_at_pt, buf, 256)
print(f'클릭 좌표의 창: "{buf.value}" (hwnd={hex(hwnd_at_pt)}, 대상={hex(hwnd)})')
if hwnd_at_pt != hwnd:
    # 자식 창일 수 있으니 최상위 부모 확인
    root = user32.GetAncestor(hwnd_at_pt, 2)  # GA_ROOT=2
    if root != hwnd:
        print('경고: 클릭 좌표에 다른 창이 있습니다. 대상 창이 가려져 있을 수 있습니다.')

# 클릭 시도
try:
    element.click()  # WM 메시지 방식 (관리자 앱이면 실패)
    print('클릭 성공 (WM 메시지)')
except Exception as e:
    if "rights" in str(e).lower() or "privilege" in str(e).lower():
        print(f'WM 클릭 불가 (관리자 권한 앱): {e}')
        print('pyautogui SendInput 방식으로 재시도...')
        pyautogui.click(center_x, center_y)
        print('pyautogui 클릭 완료')
    else:
        raise
```
